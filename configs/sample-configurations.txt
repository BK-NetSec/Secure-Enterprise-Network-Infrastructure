#==================== VRRP sample (report Appendix A) ====================
# NOTE: the report labels these "CORE-SW1/2" and VLAN 10 with VIP 192.168.10.1.
# The verified output in the test results shows VIP 192.168.10.254 on the distribution switches.
# Master
interface Vlanif10
 ip address 192.168.10.2 255.255.255.0
 vrrp vrid 10 virtual-ip 192.168.10.1
 vrrp vrid 10 priority 120
 vrrp vrid 10 preempt-mode enable
# Backup
interface Vlanif10
 ip address 192.168.10.3 255.255.255.0
 vrrp vrid 10 virtual-ip 192.168.10.1
 vrrp vrid 10 priority 100
 vrrp vrid 10 preempt-mode enable
